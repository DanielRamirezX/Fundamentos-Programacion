// Un reactivo tipico mezcla VARIOS temas en un solo problema:
// archivo + ciclo + diccionario (map) + excepciones + funcion.
#include <iostream>
#include <fstream>
#include <sstream>
#include <string>
#include <map>
#include <iomanip>
using namespace std;

// Lee un .csv y lo devuelve como diccionario producto -> stock.
map<string, int> cargar_inventario(string ruta) {
    map<string, int> inventario;
    ifstream archivo(ruta);
    if (!archivo) {
        cout << "No encontre el archivo. Revisa la ruta." << endl;
        return inventario;
    }
    string linea;
    while (getline(archivo, linea)) {
        stringstream ss(linea);
        string nombre, cantidad;
        getline(ss, nombre, ',');
        getline(ss, cantidad);
        try {
            inventario[nombre] = stoi(cantidad);
        } catch (const exception& e) {
            cout << "Cantidad invalida: " << cantidad << endl;
        }
    }
    return inventario;
}

int main() {
    map<string, int> datos = cargar_inventario("almacen.csv");
    for (auto& par : datos) {
        string estado = (par.second == 0) ? "AGOTADO" : "ok";
        cout << setw(12) << left << par.first
             << setw(3) << right << par.second
             << "  " << estado << endl;
    }
    return 0;
}

// Azucar       24  ok
// Cafe          0  AGOTADO
asi_se_ve_un_reactivo.py
# Un reactivo típico mezcla VARIOS temas en un solo problema:
# archivo + ciclo + diccionario + excepciones + función.

def cargar_inventario(ruta):
    """Lee un .csv y lo devuelve como diccionario producto -> stock."""
    inventario = {}
    try:
        with open(ruta, encoding="utf-8") as archivo:
            for linea in archivo:
                nombre, cantidad = linea.strip().split(",")
                inventario[nombre] = int(cantidad)
    except FileNotFoundError:
        print("No encontré el archivo. Revisa la ruta.")
    return inventario

datos = cargar_inventario("almacen.csv")
for producto, stock in datos.items():
    estado = "AGOTADO" if stock == 0 else "ok"
    print(f"{producto:12} {stock:3}  {estado}")

# Café           0  AGOTADO
# Azúcar        24  ok

iv.

La regla de honor.

Un examen es un pacto de honestidad. Lo que demuestres hoy tiene que ser tuyo: por eso, desde la sesión 07, uno de los cuatro criterios de evaluación es la autenticidad. Aquí está, sin letra chica.

Sí · esto se vale

  • Escribir tu código desde cero, con tu propia lógica.
  • Correr y depurar en VS Code cuantas veces necesites.
  • Consultar la documentación oficial de C++ (o de Python) si se autoriza.
  • Hacer tu prueba de escritorio en papel antes de teclear.
  • Preguntarme a mí si una instrucción no se entiende.

No · esto anula el examen

  • Copiar el código de un compañero (o dejar que copien el tuyo).
  • Traer la solución preescrita en notas, USB o chats.
  • Pegar el enunciado en una IA para que la resuelva por ti.
  • Comunicarte con alguien durante la prueba.
  • Entregar como propio un programa que no entiendes.

Sobre la inteligencia artificial

A lo largo del curso te animé a usar la IA como un maestro paciente: que te explique un error, que te muestre un ejemplo, que te enseñe lo que no entendiste en clase. Esa es su mejor cara, y la seguirás usando toda tu vida profesional.

Pero el examen es la excepción. Hoy no se trata de obtener la respuesta correcta, sino de demostrar que tú puedes producirla. Pedirle a una IA que resuelva tu examen es como pedirle a alguien que corra el maratón en tu lugar y luego colgarte la medalla: el papel dirá que cruzaste la meta, pero las piernas seguirán sin saber correr.

Por eso, durante esta prueba, no se permite el uso de IA generativa de código. No por desconfianza —sino porque el único que sale perdiendo si la usas, eres tú. Confía en lo que aprendiste. Te alcanza.

Criterios de evaluación

  1. Sin errores de sintaxis ni de ejecución — tu programa corre completo, sin reventar.
  2. Lógica correcta — resuelve de verdad lo que el problema pide, sin errores silenciosos.
  3. Funcionamiento óptimo — cumple todos los requisitos del reactivo, con código claro y comentado.
  4. Autenticidad — la solución es tuya, pensada y escrita por ti durante el examen.

Antes de empezar

Una palabra para ti, antes del primer print() de hoy.

Hace catorce semanas, muchos de ustedes nunca habían escrito una línea de código. Hoy leen un programa que mezcla archivos, diccionarios, ciclos y excepciones —y lo entienden. Eso no pasó por accidente: pasó porque vinieron, preguntaron, se equivocaron, cazaron sus propios bugs y volvieron a intentarlo. Aprender a programar es, sobre todo, aprender a no rendirse ante el error.

Este examen no es un muro al final del camino: es un espejo. Te va a devolver, con honestidad, lo que ya sabes hacer. Y si algo no sale, también está bien —porque el lunes que viene abriremos juntos cada error y lo convertiremos en el último aprendizaje del curso.

Respira. Lee con calma. Empieza por lo fácil. Haz tu prueba de escritorio. Y recuerda lo más importante de todo el semestre: el código que «funciona» también puede mentir —así que desconfía, prueba, verifica. Esa desconfianza sana es lo que te vuelve programador.

Confío en ustedes. Mucho. Ahora, a escribir.
